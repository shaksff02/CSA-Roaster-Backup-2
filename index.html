/** * CSA Duty Ledger — Web App backend (Google Apps Script) * ------------------------------------------------------ * Bind this script to the SAME Google Sheet that "Schedule", * "Form_Responses" and "Swap_Requests" tabs live in * (Extensions ▸ Apps Script from inside the Sheet). * * Deploy ▸ New deployment ▸ Web app * Execute as: Me * Who has access: Anyone * Copy the /exec URL into CONFIG.OT_API_URL in index.html * (it can stay the SAME url you already use for OT — this * script now handles OT + Swap requests + Swap approvals, * routed by the "action" field in the POST body). * * IMPORTANT: change ADMIN_PASSCODE below to your own value, * and put the EXACT same string in CONFIG.ADMIN_PASSCODE in * index.html. This is a light gate for a small internal team — * not real authentication (anyone who can read the client-side * code can see it, since there's no real server/login here). */ const SHEET_NAME = "Schedule"; const OT_SHEET_NAME = "Form_Responses"; const SWAP_SHEET_NAME = "Swap_Requests"; const ADMIN_PASSCODE = "changeme123"; // ⚠️ change me, keep in sync with index.html function doPost(e) { let result; try { const body = JSON.parse(e.postData.contents); const action = body.action || "add_ot"; // default = old OT client, backward compatible if (action === "add_ot") result = handleAddOT_(body); else if (action === "swap_request") result = handleSwapRequest_(body); else if (action === "swap_review") result = handleSwapReview_(body); else result = { ok: false, error: "unknown action: " + action }; } catch (err) { result = { ok: false, error: String(err) }; } return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { return ContentService.createTextOutput("CSA Duty Ledger backend is running."); } /* ---------------- helpers ---------------- */ function ss_() { return SpreadsheetApp.getActiveSpreadsheet(); } function sheet_(name) { const sh = ss_().getSheetByName(name); if (!sh) throw new Error('Sheet tab not found: "' + name + '". Create it first.'); return sh; } function normalizeAgentName_(name) { return String(name || "") .toLowerCase() .replace(/\./g, "") .replace(/\s+/g, " ") .trim(); } function toDateKey_(d) { // Accepts a JS Date, a sheet cell Date, or a "yyyy-mm-dd" string. if (Object.prototype.toString.call(d) === "[object Date]" && !isNaN(d)) { return Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd"); } const s = String(d || "").trim(); const m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/); if (m) return `${m[1]}-${String(m[2]).padStart(2,'0')}-${String(m[3]).padStart(2,'0')}`; const parsed = new Date(s); if (!isNaN(parsed)) return Utilities.formatDate(parsed, Session.getScriptTimeZone(), "yyyy-MM-dd"); return s; } /* ---------------- OT (existing behavior, preserved) ---------------- */ function handleAddOT_(body) { const sh = sheet_(OT_SHEET_NAME); sh.appendRow([new Date(), body.date, body.name, body.hours]); return { ok: true }; } /* ---------------- Swap: create request (Pending) ---------------- */ function handleSwapRequest_(body) { const sh = sheet_(SWAP_SHEET_NAME); const id = Utilities.getUuid(); sh.appendRow([ new Date(), // Timestamp id, // Request ID body.date, // Duty Date (yyyy-mm-dd) body.requesterName || "", body.requesterKey || "", body.requesterShift || "", body.requesterRegion || "", body.partnerName || "", body.partnerKey || "", body.partnerShift || "", "Pending", // Status "", // Reviewed At "" // Reviewed Note ]); return { ok: true, id: id }; } /* ---------------- Swap: admin approve / reject ---------------- */ function handleSwapReview_(body) { if (String(body.passcode) !== String(ADMIN_PASSCODE)) { return { ok: false, error: "invalid passcode" }; } const sh = sheet_(SWAP_SHEET_NAME); const data = sh.getDataRange().getValues(); let rowIdx = -1; for (let i = 1; i < data.length; i++) { if (String(data[i][1]) === String(body.requestId)) { rowIdx = i; break; } } if (rowIdx === -1) return { ok: false, error: "request not found" }; const row = data[rowIdx]; const dutyDateKey = toDateKey_(row[2]); const requesterKey = String(row[4] || "").trim(); const partnerKey = String(row[8] || "").trim(); if (body.decision === "approve") { applySwapToSchedule_(dutyDateKey, requesterKey, partnerKey); sh.getRange(rowIdx + 1, 11).setValue("Approved"); } else { sh.getRange(rowIdx + 1, 11).setValue("Rejected"); } sh.getRange(rowIdx + 1, 12).setValue(new Date()); sh.getRange(rowIdx + 1, 13).setValue(body.note || ""); return { ok: true }; } /** * Actually swaps the two agents' names on the Schedule tab for the * given date. The "slot" (Shift / System ID / Region) stays put — * only which agent is assigned to it changes. This mirrors exactly * what the Swap Helper UI already shows as "eligible partners". */ function applySwapToSchedule_(dateKey, requesterKey, partnerKey) { const sh = sheet_(SHEET_NAME); const data = sh.getDataRange().getValues(); const header = data[0].map(h => String(h).trim().toLowerCase()); const dateCol = header.findIndex(h => h.includes("date")); const agentCol = header.findIndex(h => h.includes("agent")); if (dateCol === -1 || agentCol === -1) throw new Error("Schedule headers not found"); let reqRow = -1, partRow = -1; for (let i = 1; i < data.length; i++) { if (toDateKey_(data[i][dateCol]) !== dateKey) continue; const key = normalizeAgentName_(data[i][agentCol]); if (key === requesterKey) reqRow = i; if (key === partnerKey) partRow = i; } if (reqRow === -1 || partRow === -1) { throw new Error("Could not find both schedule rows for " + dateKey); } const reqName = data[reqRow][agentCol]; const partName = data[partRow][agentCol]; sh.getRange(reqRow + 1, agentCol + 1).setValue(partName); sh.getRange(partRow + 1, agentCol + 1).setValue(reqName); }
